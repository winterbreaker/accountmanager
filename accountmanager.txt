local Players = game:GetService("Players")
local player = Players.LocalPlayer

-- SETTINGS
local TOTAL_SLOTS = 300
local TIMER_SECONDS = 12 * 60
local MAX_WARNINGS = 3
local GITHUB_URL = "https://github.com/USERNAME/REPOSITORY"

-- STATE
local selectedSlot = 1
local running = false
local remaining = TIMER_SECONDS
local warnings = 0
local errors = 0

local cooldownStatus = {}
local accountNames = {}

for i = 1, TOTAL_SLOTS do
    cooldownStatus[i] = false
    accountNames[i] = ""
end

--==================================================
-- GUI
--==================================================

local gui = Instance.new("ScreenGui")
gui.Name = "MyAccountManager"
gui.ResetOnSpawn = false
gui.Parent = player:WaitForChild("PlayerGui")

local main = Instance.new("Frame")
main.Name = "Main"
main.Size = UDim2.fromOffset(380, 610)
main.Position = UDim2.new(0.5, -190, 0.5, -305)
main.BackgroundColor3 = Color3.fromRGB(32, 29, 26)
main.BorderSizePixel = 0
main.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 12)
corner.Parent = main

local function createLabel(name, text, y, height)
    local label = Instance.new("TextLabel")
    label.Name = name
    label.Size = UDim2.new(1, -20, 0, height)
    label.Position = UDim2.fromOffset(10, y)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
    label.Parent = main
    return label
end

local function createButton(name, text, x, y, width, height)
    local button = Instance.new("TextButton")
    button.Name = name
    button.Size = UDim2.fromOffset(width, height)
    button.Position = UDim2.fromOffset(x, y)
    button.BackgroundColor3 = Color3.fromRGB(105, 75, 48)
    button.TextColor3 = Color3.new(1, 1, 1)
    button.Text = text
    button.TextScaled = true
    button.Font = Enum.Font.GothamBold
    button.Parent = main

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 7)
    c.Parent = button

    return button
end

local title = createLabel(
    "Title",
    "MY ACCOUNT MANAGER",
    10,
    35
)

local subtitle = createLabel(
    "Subtitle",
    "300 ACCOUNT SLOTS",
    48,
    25
)

local timerLabel = createLabel(
    "Timer",
    "12:00",
    82,
    55
)

local statusLabel = createLabel(
    "Status",
    "STATUS: STOPPED",
    140,
    27
)

local selectedLabel = createLabel(
    "Selected",
    "ACCOUNT: 1 / 300",
    178,
    27
)

local cooldownLabel = createLabel(
    "Cooldown",
    "COOLDOWN: NOT IN CD",
    210,
    27
)

--==================================================
-- ACCOUNT SLOTS
--==================================================

local previousButton = createButton(
    "Previous",
    "PREVIOUS",
    15, 250, 165, 35
)

local nextButton = createButton(
    "Next",
    "NEXT",
    200, 250, 165, 35
)

local cooldownButton = createButton(
    "CooldownToggle",
    "TOGGLE COOLDOWN",
    15, 295, 350, 35
)

local function updateAccount()
    selectedLabel.Text =
        "ACCOUNT: " .. selectedSlot .. " / " .. TOTAL_SLOTS

    if cooldownStatus[selectedSlot] then
        cooldownLabel.Text = "COOLDOWN: IN CD"
        cooldownLabel.TextColor3 =
            Color3.fromRGB(255, 180, 65)
    else
        cooldownLabel.Text = "COOLDOWN: NOT IN CD"
        cooldownLabel.TextColor3 =
            Color3.fromRGB(100, 255, 120)
    end
end

previousButton.MouseButton1Click:Connect(function()
    selectedSlot = math.max(1, selectedSlot - 1)
    updateAccount()
end)

nextButton.MouseButton1Click:Connect(function()
    selectedSlot = math.min(TOTAL_SLOTS, selectedSlot + 1)
    updateAccount()
end)

cooldownButton.MouseButton1Click:Connect(function()
    cooldownStatus[selectedSlot] =
        not cooldownStatus[selectedSlot]

    updateAccount()
end)

--==================================================
-- TIMER
--==================================================

local startButton = createButton(
    "Start",
    "START",
    15, 340, 165, 40
)

local stopButton = createButton(
    "Stop",
    "STOP",
    200, 340, 165, 40
)

local function updateTimer()
    local minutes = math.floor(remaining / 60)
    local seconds = remaining % 60

    timerLabel.Text = string.format(
        "%02d:%02d",
        minutes,
        seconds
    )
end

startButton.MouseButton1Click:Connect(function()
    if running then return end

    running = true
    statusLabel.Text = "STATUS: RUNNING"
end)

stopButton.MouseButton1Click:Connect(function()
    running = false
    statusLabel.Text = "STATUS: STOPPED"
end)

task.spawn(function()
    while gui.Parent do
        task.wait(1)

        if running then
            remaining -= 1

            if remaining <= 0 then
                remaining = TIMER_SECONDS
                statusLabel.Text = "STATUS: TIMER RESET"
            end

            updateTimer()
        end
    end
end)

--==================================================
-- SAFETY PROTECTION
--==================================================

local safetyTitle = createLabel(
    "SafetyTitle",
    "ANTI-BAN & WARNING PROTECTION",
    505,
    25
)

local safetyStatus = createLabel(
    "SafetyStatus",
    "SAFETY: ACTIVE",
    533,
    22
)

local warningLabel = createLabel(
    "Warnings",
    "WARNINGS: 0 | ERRORS: 0",
    556,
    22
)

local function updateWarnings()
    warningLabel.Text =
        "WARNINGS: " .. warnings ..
        " | ERRORS: " .. errors
end

local function AddWarning(message)
    warnings += 1
    updateWarnings()

    safetyStatus.Text = "WARNING: " .. tostring(message)
    safetyStatus.TextColor3 =
        Color3.fromRGB(255, 190, 65)

    if warnings >= MAX_WARNINGS then
        running = false
        statusLabel.Text = "STATUS: SAFETY STOP"
    end
end

local function AddError(message)
    errors += 1
    running = false

    updateWarnings()

    statusLabel.Text = "STATUS: ERROR STOP"
    safetyStatus.Text = "ERROR: " .. tostring(message)
    safetyStatus.TextColor3 =
        Color3.fromRGB(255, 75, 75)
end

local resetButton = createButton(
    "ResetSafety",
    "RESET SAFETY COUNTERS",
    15, 580, 350, 25
)

resetButton.MouseButton1Click:Connect(function()
    warnings = 0
    errors = 0

    updateWarnings()

    safetyStatus.Text = "SAFETY: ACTIVE"
    safetyStatus.TextColor3 =
        Color3.fromRGB(100, 255, 120)
end)

--==================================================
-- INITIALIZATION
--==================================================

updateAccount()
updateTimer()
updateWarnings()
